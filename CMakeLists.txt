cmake_minimum_required(VERSION 3.28)
project(QCopilotsWebSearch VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(QCWS_BUILD_TESTING "Build deterministic tests" ON)
option(QCWS_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
include(cmake/Dependencies.cmake)
add_library(qcws_core STATIC src/network.cpp src/cache.cpp src/config.cpp src/html.cpp src/service.cpp src/protocol.cpp src/server.cpp)
set(_schemas config/runtime.json config/runtime.schema.json contracts/v1/web_search.schema.json contracts/v1/web_open.schema.json
  contracts/v1/web_find.schema.json contracts/v1/web_search_status.schema.json)
foreach(schema IN LISTS _schemas)
  get_filename_component(alias "${schema}" NAME)
  set_source_files_properties("${schema}" PROPERTIES QT_RESOURCE_ALIAS "${alias}")
endforeach()
qt_add_resources(qcws_core schemas PREFIX /schemas FILES ${_schemas})
target_include_directories(qcws_core PUBLIC include)
target_link_libraries(qcws_core PUBLIC Qt6::Core Qt6::Network Qt6::Sql nlohmann_json::nlohmann_json
  PRIVATE CURL::libcurl lexbor_static nlohmann_json_schema_validator::validator httplib::httplib)
target_compile_definitions(qcws_core PRIVATE CPPHTTPLIB_HEADER_MAX_LENGTH=16384 CPPHTTPLIB_HEADER_MAX_COUNT=64 CPPHTTPLIB_REQUEST_URI_MAX_LENGTH=8192)
add_executable(qcopilots-websearch app/main.cpp)
target_link_libraries(qcopilots-websearch PRIVATE qcws_core)
if(WIN32)
  target_link_libraries(qcopilots-websearch PRIVATE advapi32)
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION bin)
  include(InstallRequiredSystemLibraries)
endif()
include(GNUInstallDirs)
install(TARGETS qcopilots-websearch RUNTIME DESTINATION bin)
install(DIRECTORY contracts/ DESTINATION share/qcopilots-websearch/contracts)
install(FILES config/runtime.json config/runtime.schema.json DESTINATION share/qcopilots-websearch/config)
qt_generate_deploy_app_script(TARGET qcopilots-websearch OUTPUT_SCRIPT _deploy_script NO_UNSUPPORTED_PLATFORM_ERROR
  DEPLOY_TOOL_OPTIONS --no-translations --no-compiler-runtime --exclude-plugins qsqlibase,qsqlmimer,qsqlmysql,qsqloci,qsqlodbc,qsqlpsql)
install(SCRIPT ${_deploy_script})
include(cmake/Packaging.cmake)
if(MSVC)
  target_compile_options(qcws_core PRIVATE /W4 /utf-8 /permissive-)
else()
  target_compile_options(qcws_core PRIVATE -Wall -Wextra -Wpedantic)
  if(QCWS_SANITIZERS)
    target_compile_options(qcws_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(qcws_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
if(QCWS_BUILD_TESTING)
  enable_testing()
  add_executable(qcws_tests tests/test_core.cpp)
  target_link_libraries(qcws_tests PRIVATE qcws_core httplib::httplib)
  if(MSVC)
    target_compile_options(qcws_tests PRIVATE /utf-8)
  endif()
  add_test(NAME core COMMAND qcws_tests)
  set_tests_properties(core PROPERTIES TIMEOUT 60)
endif()
