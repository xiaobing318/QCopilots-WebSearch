cmake_minimum_required(VERSION 3.28)
set(QCWS_VERSION "0.5.0" CACHE STRING "Three-part release or isolated installer-test version")
if(NOT QCWS_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
  message(FATAL_ERROR "QCWS_VERSION must contain three numeric components")
endif()
project(QCopilotsWebSearch VERSION ${QCWS_VERSION} LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(QCWS_BUILD_TESTING "Build deterministic tests" ON)
option(QCWS_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(QCWS_FUZZING "Build bounded Clang libFuzzer targets" OFF)
if(QCWS_SANITIZERS)
  if(MSVC)
    message(FATAL_ERROR "Combined ASan/UBSan checks require the Linux quality build")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(QCWS_FUZZING AND (NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR NOT QCWS_SANITIZERS))
  message(FATAL_ERROR "Fuzzing requires Linux, Clang and QCWS_SANITIZERS")
endif()
include(cmake/Dependencies.cmake)
add_library(qcws_core STATIC src/core/network.cpp src/core/cache.cpp src/core/config.cpp src/core/html.cpp src/core/service.cpp src/server/protocol.cpp src/server/server.cpp src/server/stdio.cpp src/server/stdio_output.cpp
  src/core/settings.cpp src/core/client_registry.cpp src/server/admin_api.cpp src/server/admin_sessions.cpp
  src/server/admin_client.cpp src/server/static_assets.cpp
  src/providers/registry.cpp src/providers/searxng.cpp src/providers/auto_provider.cpp
  src/providers/pacing.cpp src/providers/duckduckgo.cpp src/providers/bing.cpp
  src/providers/serp_parsers.cpp src/providers/brave.cpp src/providers/tavily.cpp
  src/providers/exa.cpp src/providers/perplexity.cpp src/providers/deepseek.cpp
  src/providers/locale_map.cpp)
include(cmake/WebUi.cmake)
set(_schemas config/runtime.json config/runtime.schema.json contracts/v1/web_search.schema.json contracts/v1/web_open.schema.json
  contracts/v1/web_find.schema.json contracts/v1/web_search_status.schema.json contracts/v1/tools.json
  config/settings.json config/settings.schema.json config/secrets.schema.json config/client-registry.schema.json
  contracts/admin/v1/settings-update.schema.json contracts/admin/v1/settings-view.schema.json)
foreach(schema IN LISTS _schemas)
  get_filename_component(alias "${schema}" NAME)
  set_source_files_properties("${schema}" PROPERTIES QT_RESOURCE_ALIAS "${alias}")
endforeach()
qt_add_resources(qcws_core schemas PREFIX /schemas FILES ${_schemas})
set_source_files_properties(resources/websearch.svg PROPERTIES QT_RESOURCE_ALIAS websearch.svg)
qt_add_resources(qcws_core branding PREFIX /branding FILES resources/websearch.svg)
target_include_directories(qcws_core PUBLIC include)
target_compile_definitions(qcws_core PUBLIC QCWS_VERSION="${PROJECT_VERSION}")
target_link_libraries(qcws_core PUBLIC Qt6::Core Qt6::Network Qt6::Sql nlohmann_json::nlohmann_json
  PRIVATE CURL::libcurl lexbor_static nlohmann_json_schema_validator::validator httplib::httplib)
target_compile_definitions(qcws_core PUBLIC CPPHTTPLIB_HEADER_MAX_LENGTH=16384 CPPHTTPLIB_HEADER_MAX_COUNT=64 CPPHTTPLIB_REQUEST_URI_MAX_LENGTH=8192)
add_executable(qcopilots-websearch app/main.cpp)
target_link_libraries(qcopilots-websearch PRIVATE qcws_core)
if(UNIX)
  set_target_properties(qcopilots-websearch PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
endif()
if(WIN32)
  target_sources(qcws_core PRIVATE src/platform/windows.cpp)
  target_link_libraries(qcws_core PUBLIC advapi32 shell32)
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION bin)
  include(InstallRequiredSystemLibraries)
endif()
include(GNUInstallDirs)
include(cmake/Install.cmake)
include(cmake/Packaging.cmake)
if(MSVC)
  target_compile_options(qcws_core PRIVATE /W4 /utf-8 /permissive-)
else()
  target_compile_options(qcws_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(QCWS_FUZZING)
  target_compile_options(qcws_core PRIVATE -fsanitize=fuzzer-no-link)
  target_compile_options(lexbor_static PRIVATE -fsanitize=fuzzer-no-link)
  set(_fuzz_targets html url jsonrpc)
  if(EXISTS "${PROJECT_SOURCE_DIR}/tests/fuzz/serp.cpp")
    list(APPEND _fuzz_targets serp)
  endif()
  foreach(kind IN LISTS _fuzz_targets)
    add_executable(qcws_${kind}_fuzz tests/fuzz/${kind}.cpp tests/fuzz/qt_init.cpp)
    target_link_libraries(qcws_${kind}_fuzz PRIVATE qcws_core)
    target_compile_options(qcws_${kind}_fuzz PRIVATE -fsanitize=fuzzer-no-link)
    target_link_options(qcws_${kind}_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
    if(QCWS_BUILD_TESTING)
      file(GLOB seeds CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/tests/fuzz/corpus/${kind}/*")
      if(NOT seeds)
        message(FATAL_ERROR "Missing ${kind} fuzz seeds")
      endif()
      add_test(NAME fuzz-${kind}-seeds COMMAND qcws_${kind}_fuzz -runs=1 -timeout=5 -rss_limit_mb=1024 ${seeds})
      set_tests_properties(fuzz-${kind}-seeds PROPERTIES TIMEOUT 45)
    endif()
  endforeach()
endif()
if(QCWS_BUILD_TESTING)
  set_property(SOURCE src/server/server.cpp APPEND PROPERTY COMPILE_DEFINITIONS QCWS_TEST_LISTENER_ACCESS)
  enable_testing()
  add_subdirectory(tests)
endif()
